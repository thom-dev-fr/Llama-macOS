cmake_minimum_required(VERSION 3.24)
project(LlamaMobile LANGUAGES C CXX)
# Configured by scripts/build-ios-engine.sh, which pins the llama.cpp version.
foreach(var LLAMA_SOURCE LLAMA_BUILD_NUMBER LLAMA_BUILD_COMMIT)
  if (NOT DEFINED ${var})
    message(FATAL_ERROR "${var} is not set; build through scripts/build-ios-engine.sh")
  endif()
endforeach()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_MACOSX_BUNDLE OFF)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_COMMON ON CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TOOLS ON CACHE BOOL "" FORCE)
set(LLAMA_BUILD_SERVER ON CACHE BOOL "" FORCE)
set(LLAMA_BUILD_APP OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LLAMA_OPENSSL OFF CACHE BOOL "" FORCE)
# iOS apps cannot spawn processes; MobileServer.cpp stands in for the router mode that needs them.
set(LLAMA_SUBPROCESS OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_METAL_EMBED_LIBRARY ON CACHE BOOL "" FORCE)
add_subdirectory("${LLAMA_SOURCE}" llama)
add_library(mobile-server STATIC MobileServer.cpp "${LLAMA_SOURCE}/tools/server/server-http.cpp")
target_include_directories(mobile-server PRIVATE "${LLAMA_SOURCE}/tools/server" "${LLAMA_SOURCE}/tools/mtmd" "${LLAMA_SOURCE}")
target_link_libraries(mobile-server PUBLIC server-context llama-ui cpp-httplib)
